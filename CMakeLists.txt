cmake_minimum_required(VERSION 3.31)
project(muser VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(USE_DESKTOP "Build the SDL desktop simulator" ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(lvgl EXCLUDE_FROM_ALL URL https://github.com/lvgl/lvgl/archive/refs/tags/v9.5.0.tar.gz)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.cache/lvgl/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_LVGL "${CMAKE_CURRENT_SOURCE_DIR}/.cache/lvgl")
endif()
FetchContent_MakeAvailable(lvgl)
find_package(Freetype REQUIRED)
find_package(PNG REQUIRED)
find_package(Threads REQUIRED)
target_link_libraries(lvgl PUBLIC Freetype::Freetype Threads::Threads)
target_compile_definitions(lvgl PUBLIC USE_DESKTOP=$<BOOL:${USE_DESKTOP}>)
add_executable(muser src/main.cpp src/bridge.cpp src/input.cpp vendor/muse_pixel.c)
target_include_directories(muser PRIVATE vendor src)
target_link_libraries(muser PRIVATE lvgl PNG::PNG ${CMAKE_DL_LIBS} m)
target_compile_definitions(muser PRIVATE MUSER_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" MUSER_VERSION="${PROJECT_VERSION}")
target_compile_options(muser PRIVATE -Wall -Wextra -Wpedantic)
if(USE_DESKTOP)
    find_package(SDL2 CONFIG REQUIRED)
    target_link_libraries(muser PRIVATE SDL2::SDL2)
endif()
install(TARGETS muser RUNTIME DESTINATION lib/muser)
install(PROGRAMS packaging/muser DESTINATION bin)
install(DIRECTORY backend vendor DESTINATION share/muser
    PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE PATTERN "*.c" EXCLUDE PATTERN "*.h" EXCLUDE PATTERN "*.hpp" EXCLUDE)
install(DIRECTORY assets DESTINATION share/muser)
install(FILES packaging/muser.desktop DESTINATION share/APPLaunch/applications)
install(FILES assets/images/muser.png DESTINATION share/APPLaunch/share/images)
install(FILES README.md THIRD_PARTY_LICENSES.md NOTICE LICENSE config.example.json DESTINATION share/doc/muser)
if(NOT USE_DESKTOP)
    set(CPACK_GENERATOR DEB)
    set(CPACK_PACKAGE_NAME muser)
    set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
    set(CPACK_PACKAGE_CONTACT "Muser community")
    set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Muser community")
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
    set(CPACK_DEBIAN_PACKAGE_SECTION utils)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6 (>= 2.38), libstdc++6 (>= 13.2), libgcc-s1, libfreetype6, libpng16-16, zlib1g, python3 (>= 3.9), python3-cryptography, python3-cffi, python3-dbus, python3-gi, libportaudio2, bluez, ca-certificates")
    set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
    set(CPACK_OUTPUT_FILE_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/dist")
    set(CPACK_PACKAGE_FILE_NAME "muser_${PROJECT_VERSION}_arm64")
    include(CPack)
endif()
